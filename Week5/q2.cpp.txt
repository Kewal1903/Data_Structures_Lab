#include<iostream>
using namespace std;
struct Stack{
        int stack[100];
        int top;
        int size;
        bool isValid;
};
bool isFull(Stack *inst){
        if (inst -> top == inst -> size){
                return true;}
        else{
                return false;}
}
bool isEmpty(Stack *inst){
        if (inst -> top == 0){
                return true;}
        else{
                return false;}
}
void push(Stack *inst, int element){
        if (isFull(inst)){
                cout<<"\n Overflow, stack is full. \n";
                return;}
        inst -> stack[inst -> top] = element;
        inst -> top++;
}
